CREATE OR REPLACE PROCEDURE public.addvehicle(OUT p_id integer, IN p_make character varying, IN p_model character varying, IN p_year integer, IN p_vin character varying)
 LANGUAGE plpgsql
AS $procedure$
BEGIN
    INSERT INTO vehicle(
        make,
        model,
        year,
        vin,
        created_at,
        updated_at
    )
    VALUES(
        p_make,
        p_model,
        p_year,
        p_vin,
        CURRENT_TIMESTAMP,
        CURRENT_TIMESTAMP
    )
    RETURNING vehicle.id INTO p_id;
END;
$procedure$

CREATE OR REPLACE PROCEDURE public.searchvehicle(IN p_make character varying, OUT result character varying)
 LANGUAGE plpgsql
AS $procedure$
declare
    vehicle_count integer;
BEGIN
    select count(*) into vehicle_count from public.vehicle where make = p_make;
    IF vehicle_count = 0 THEN
        
        raise EXCEPTION 'No vehicles found for the specified make: %', p_make;   
    ELSIF vehicle_count = 1 THEN
        select make || ' ' || model into result from public.vehicle where make = p_make;
    ELSE
        select string_agg(make || ' ' || model, ', ') into result from public.vehicle where make = p_make;
    END IF;

END;
$procedure$

CREATE OR REPLACE FUNCTION count_vehicles_by_make(p_make character varying)
RETURNS int AS $$
BEGIN
    declare
        vehicle_count integer;
    begin
        select count(*) into vehicle_count from public.vehicle where make = p_make;
        return vehicle_count;
    end;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION public.vehicleinfo(p_make character varying)
RETURNS TABLE (v_make character varying, v_model character varying, v_year integer, v_vin character varying) AS $$
BEGIN
    RETURN QUERY
    SELECT make, model, year, vin
    FROM public.vehicle
    WHERE make = p_make;
END;
$$ LANGUAGE plpgsql;

CREATE SEQUENCE Audit_sequence start with 1000 INCREMENT BY 1 MINVALUE 1000  No MAXVALUE;
CREATE OR REPLACE FUNCTION public.audittrigger()
RETURNS TRIGGER AS $$
BEGIN
   INSERT INTO public.vehicle_audit (audit_id,vehicle_id,make,model,year,vin,created_at,updated_at)
   VALUES (nextval('Audit_sequence'), NEW.vehicle_id, NEW.make, NEW.model, NEW.year, NEW.vin, NEW.created_at, NEW.updated_at);
   RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER VehicleTableTrigger
BEFORE INSERT OR UPDATE OR DELETE
ON vehicle
FOR EACH ROW
EXECUTE PROCEDURE public.audittrigger();
