Base tables
CREATE TABLE vehicle (
    vehicle_id SERIAL PRIMARY KEY,
    vin VARCHAR(30) UNIQUE NOT NULL,
    model VARCHAR(50),
    fuel_type VARCHAR(20),
    region VARCHAR(50),
    price NUMERIC(12,2)

CREATE TABLE sales (
    sale_id SERIAL PRIMARY KEY,
    vehicle_id INT REFERENCES vehicle(vehicle_id),
    customer_name VARCHAR(100),
    city VARCHAR(50),
    sale_date DATE,
    quantity INT DEFAULT 1,
    sale_amount NUMERIC(12,2)
);

    service_id SERIAL PRIMARY KEY,

    service_date DATE,
    service_type VARCHAR(50),
    service_cost NUMERIC(10,2)
);

CREATE TABLE vehicle_audit (
    audit_id SERIAL PRIMARY KEY,
    vehicle_id INT,
    old_price NUMERIC(12,2),
    new_price NUMERIC(12,2),
    changed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

Advanced Challenge — Top 3 Models by Region

Use a CTE + window function.

Requirement:

Find the top 3 highest-revenue BMW models in each city/region.

Conceptual solution:

WITH model_sales AS
(
    SELECT
        s.city,
        v.model,
        SUM(s.sale_amount) AS revenue
    FROM sales s
    JOIN vehicle v
        ON s.vehicle_id = v.vehicle_id
    GROUP BY
        s.city,
        v.model
),
ranking AS
(
    SELECT
        *,
        DENSE_RANK() OVER (
            PARTITION BY city
            ORDER BY revenue DESC
        ) AS ranking
    FROM model_sales
)

SELECT *
FROM ranking
WHERE ranking <= 3;

This combines JOIN + GROUP BY + CTE + DENSE_RANK in one realistic analytical query.
